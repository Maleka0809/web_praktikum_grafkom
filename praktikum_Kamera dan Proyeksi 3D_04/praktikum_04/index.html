Penjelasan Detail Konsep & Jawaban Challenge Praktikum 04

1. Benang Merah Pipeline 3D (Vertex ke Layar)

Dalam grafika komputer 3D, sebuah vertex atau titik pada objek melewati tahapan transformasi (Pipeline) berurutan sebelum digambar di layar:

  1. Local Space: Koordinat asli objek (misal: titik-titik pada kubus).
  2. Model Matrix: Memindahkan titik dari Local ke World Space (Mentranslasi, merotasi, dan menskala objek ke posisinya di dunia 3D).
  3. View Matrix: Memindahkan titik dari World ke View Space (Mengubah sudut pandang agar relatif terhadap posisi dan arah kamera).
  4. Projection Matrix: Memindahkan titik dari View ke Clip Space (Menerapkan efek perspektif atau ortografis dan memotong objek di luar jangkauan pandang).
  5. Perspective Divide & NDC: Pembagian dengan koordinat homogen (W) untuk menghasilkan Normalized Device Coordinates (NDC).
  6. Viewport & Screen: Memetakan koordinat NDC ke resolusi layar monitor (pixel).

Pada Vertex Shader, implementasinya menggunakan perkalian matriks (Projection × View × Model × Posisi Lokal):

#version 300 es
in vec3 a_position;
in vec3 a_color;

uniform mat4 u_model;
uniform mat4 u_view;
uniform mat4 u_projection;

out vec3 v_color;

void main() {
  gl_Position = u_projection * u_view * u_model * vec4(a_position, 1.0);
  v_color = a_color;
}

2. Mengapa Menggunakan Matrix 4x4?

Koordinat 3D memiliki nilai X, Y, dan Z. Namun, transformasi linier 3x3 hanya bisa melakukan Rotasi dan Skala, tidak bisa melakukan Translasi (perpindahan). Dengan menggunakan Homogeneous Coordinate (X, Y, Z, 1) dan Matriks 4x4, kita dapat menggabungkan operasi Translasi, Rotasi, Skala, View, dan Proyeksi sekaligus di dalam satu matriks perkalian.

3. Camera (Posisi, Target, Up Vector & LookAt)

Kamera di ruang 3D tidak benar-benar bergerak. Secara matematis, View Matrix memindahkan seluruh dunia ke arah yang berlawanan dengan pergerakan kamera (Inverse Transform). Tiga komponen utama kamera:

  • Position: Koordinat lokasi kamera di dunia (misal: [0.0, 1.5, 4.0]).
  • Target: Titik yang dilihat oleh kamera (misal: [0.0, 0.0, 0.0]).
  • Up Vector: Arah "atas" pada dunia (biasanya [0.0, 1.0, 0.0]).

Basis vektor kamera dihitung dengan Cross Product. Forward didapat dari (Target - Position). Right didapat dari (Forward × Up). Corrected Up didapat dari (Right × Forward).

// Implementasi LookAt (View Matrix)
lookAt(position, target, up) {
  const forward = Vec3.normalize(Vec3.subtract(target, position));
  const right = Vec3.normalize(Vec3.cross(forward, up));
  const correctedUp = Vec3.cross(right, forward);

  return new Float32Array([
     right[0],       correctedUp[0],      -forward[0],       0,
     right[1],       correctedUp[1],      -forward[1],       0,
     right[2],       correctedUp[2],      -forward[2],       0,
    -Vec3.dot(right, position), -Vec3.dot(correctedUp, position), Vec3.dot(forward, position), 1
  ]);
}

4. Perspective vs Orthographic Projection

Projection Matrix menentukan bagaimana bentuk 3D dipetakan ke bidang 2D layar.

  • Perspective: Semakin jauh objek, semakin kecil tampilannya (seperti mata manusia). Menggunakan parameter FOV (Field of View) dan Aspect Ratio. Sering digunakan pada Game 3D.
  • Orthographic: Objek tetap ukurannya tidak peduli seberapa jauh jaraknya dari kamera. Menggunakan batasan kotak (left, right, bottom, top). Sering digunakan pada CAD, arsitektur, dan peta.
// Implementasi Projection Matrix
function createProjectionMatrix() {
  const aspect = canvas.width / canvas.height;
  if (projectionState.mode === "perspective") {
    // fov, aspect, near, far
    return Mat4.perspective(degToRad(projectionState.fov), aspect, projectionState.near, projectionState.far);
  } else {
    // left, right, bottom, top, near, far
    const size = 2.0;
    return Mat4.orthographic(-size * aspect, size * aspect, -size, size, projectionState.near, projectionState.far);
  }
}

5. Depth Buffer dan Depth Test

Tanpa Depth Test, poligon yang dirender belakangan (Draw Order) akan digambar di atas poligon yang dirender duluan, meskipun posisi aslinya berada di belakang (tertutup). Depth Test menyimpan jarak tiap piksel (Z-Buffer) ke kamera. Hanya piksel dengan nilai depth lebih kecil (lebih dekat ke kamera) yang akan digambar ke layar.

// Mengaktifkan dan Membersihkan Depth Buffer per Frame
gl.enable(gl.DEPTH_TEST);
gl.clear(gl.COLOR_BUFFER_BIT | gl.DEPTH_BUFFER_BIT);

Jawaban Lengkap Challenge Praktikum 04

Challenge A: Orbit Camera

Untuk membuat kamera bergerak melingkar mengelilingi titik pusat (target 0,0,0), kita menerapkan konsep lingkaran dari fungsi Sinus dan Cosinus. Posisi Y dapat dibuat statis, sementara X dan Z berubah berdasarkan pertambahan waktu dt.

// Modifikasi pada fungsi updateCamera()
let angle = 0;
const orbitRadius = 4.0;
const orbitSpeed = 1.0;

function updateOrbitCamera(dt) {
  angle += orbitSpeed * dt;
  
  // Mengubah posisi X dan Z berdasarkan fungsi trigonometri
  camera.position[0] = Math.cos(angle) * orbitRadius;
  camera.position[2] = Math.sin(angle) * orbitRadius;
  
  // Target tetap terkunci di origin
  camera.target = [0.0, 0.0, 0.0];
}

Challenge B: Camera Height Control

Menambahkan kemampuan menaikkan dan menurunkan kamera pada sumbu Y menggunakan tombol khusus (PageUp untuk naik, PageDown untuk turun).

// State-based keyboard controls dalam updateCamera()
const heightSpeed = 3.0;

function updateCameraHeight(dt) {
  if (keys["pageup"]) {
    camera.position[1] += heightSpeed * dt;
  }
  if (keys["pagedown"]) {
    camera.position[1] -= heightSpeed * dt;
  }
}

Challenge C: Target Control

Memindahkan pandangan kamera (Target) dengan posisi kamera tetap diam. Ini akan mengubah basis orientasi View Matrix, sehingga kamera tampak "menengok" ke arah yang berbeda.

// Mengontrol target fokus kamera
const targetSpeed = 2.0;

function updateTarget(dt) {
  if (keys["j"]) camera.target[0] -= targetSpeed * dt; // Kiri
  if (keys["l"]) camera.target[0] += targetSpeed * dt; // Kanan
  if (keys["i"]) camera.target[1] += targetSpeed * dt; // Atas
  if (keys["k"]) camera.target[1] -= targetSpeed * dt; // Bawah
}

Challenge D: Projection Comparison Split Mode

Untuk membandingkan efek secara langsung, viewport WebGL dibagi dua secara horizontal. Setengah kanvas menggunakan proyeksi Perspektif, setengahnya lagi menggunakan proyeksi Ortografis pada frame yang sama.

// Render Loop Modifikasi
function renderSplitMode() {
  gl.clear(gl.COLOR_BUFFER_BIT | gl.DEPTH_BUFFER_BIT);
  
  const aspect = (canvas.width / 2) / canvas.height;
  const viewMatrix = Mat4.lookAt(camera.position, camera.target, camera.up);

  // --- Kiri: Perspektif ---
  gl.viewport(0, 0, canvas.width / 2, canvas.height);
  const projPersp = Mat4.perspective(degToRad(60), aspect, 0.1, 100);
  drawCube(model, viewMatrix, projPersp);

  // --- Kanan: Ortografis ---
  gl.viewport(canvas.width / 2, 0, canvas.width / 2, canvas.height);
  const size = 2.0;
  const projOrtho = Mat4.orthographic(-size * aspect, size * aspect, -size, size, 0.1, 100);
  drawCube(model, viewMatrix, projOrtho);
}

Challenge E: Multiple Cube Depth Test

Menampilkan tiga objek pada kedalaman (Z) yang berbeda menggunakan transformasi Model Matrix untuk menguji apakah Z-Buffer (Depth Test) bekerja dengan benar saat posisi objek saling tumpang tindih secara visual.

// Menyiapkan Model Matriks berbeda per objek
function renderMultipleCubes() {
  const viewMatrix = Mat4.lookAt(camera.position, camera.target, camera.up);
  const projMatrix = createProjectionMatrix();

  // Kubus 1 (Belakang, Biru/Hitam)
  let model1 = Mat4.multiply(Mat4.translation(-1.5, 0.0, -3.0), Mat4.rotationY(degToRad(45)));
  drawCube(model1, viewMatrix, projMatrix);

  // Kubus 2 (Tengah, Standar)
  let model2 = Mat4.translation(0.0, 0.0, 0.0);
  drawCube(model2, viewMatrix, projMatrix);

  // Kubus 3 (Depan, Merah/Kuning)
  let model3 = Mat4.multiply(Mat4.translation(1.5, 0.0, 3.0), Mat4.rotationX(degToRad(30)));
  drawCube(model3, viewMatrix, projMatrix);
}

Challenge F: FOV Preset

Mengubah sudut Field of View secara instan antara pandangan zoom (sempit), normal, dan lebar (wide) menggunakan event berbasis tombol (Keydown event, tidak bergantung pada deltatime).

window.addEventListener("keydown", (event) => {
  if (event.key === "1") {
    projectionState.fov = 35; // Zoom in / Sempit
  }
  if (event.key === "2") {
    projectionState.fov = 60; // Normal
  }
  if (event.key === "3") {
    projectionState.fov = 90; // Wide / Lebar
  }
});
